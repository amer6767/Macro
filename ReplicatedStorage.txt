local Executor = {}

-- ── Function Detection ──────────────────────────────────────────────────────
-- Modern approach: try to reference the global directly inside pcall.
-- Avoids getfenv() which is unreliable on many modern executors.

local _funcCache = {}

function Executor.has(funcName)
    if _funcCache[funcName] ~= nil then
        return _funcCache[funcName]
    end

    local found = false

    -- Method 1: Direct global lookup (works on most modern executors)
    local ok1, result1 = pcall(function()
        local fn = rawget(_G, funcName)
        if type(fn) == "function" then return true end
        return false
    end)
    if ok1 and result1 then found = true end

    -- Method 2: Try getgenv() if available (UNC standard location)
    if not found then
        local ok2, result2 = pcall(function()
            local genv = getgenv and getgenv()
            if genv and type(genv[funcName]) == "function" then return true end
            return false
        end)
        if ok2 and result2 then found = true end
    end

    -- Method 3: getfenv(0) as last resort
    if not found then
        local ok3, result3 = pcall(function()
            local env = getfenv(0)
            if env and type(env[funcName]) == "function" then return true end
            return false
        end)
        if ok3 and result3 then found = true end
    end

    -- Method 4: Direct pcall on the name (handles identityless globals)
    if not found then
        local ok4, result4 = pcall(function()
            return type(loadstring("return type(" .. funcName .. ")")() == "function")
        end)
        if ok4 and result4 then found = true end
    end

    _funcCache[funcName] = found
    return found
end

-- ── Safe Function Resolve ───────────────────────────────────────────────────
-- Finds the actual function reference from any available environment

function Executor.resolve(funcName)
    -- Direct globals
    local ok1, fn1 = pcall(function() return rawget(_G, funcName) end)
    if ok1 and type(fn1) == "function" then return fn1 end

    -- getgenv
    local ok2, fn2 = pcall(function()
        if getgenv then return getgenv()[funcName] end
    end)
    if ok2 and type(fn2) == "function" then return fn2 end

    -- getfenv(0)
    local ok3, fn3 = pcall(function() return getfenv(0)[funcName] end)
    if ok3 and type(fn3) == "function" then return fn3 end

    return nil
end

-- ── Safe Call ───────────────────────────────────────────────────────────────

function Executor.safeCall(funcName, ...)
    local fn = Executor.resolve(funcName)
    if not fn then return false, "Function '" .. funcName .. "' not found" end
    local ok, result = pcall(fn, ...)
    return ok, result
end

-- ── UNC Environment Detection ───────────────────────────────────────────────

function Executor.detectCapabilities()
    -- Clear cache to re-detect
    _funcCache = {}

    local funcsToCheck = {
        "getgenv", "getrenv", "getsenv", "getgc", "getreg",
        "decompile", "getscriptclosure", "getscriptbytecode", "getscripthash",
        "getloadedmodules", "getrunningscripts", "saveinstance",
        "setclipboard", "firesignal", "hookfunction",
        "iscclosure", "newcclosure",
        -- Debug/analysis functions
        "getupvalues", "getconstants", "getprotos",
        "debug.getupvalue", "getupvalue",
    }

    local caps = {}
    for _, name in ipairs(funcsToCheck) do
        local cleanName = name:gsub("%.", "_")
        caps["has_" .. cleanName] = Executor.has(name)
    end

    -- Special check for Drawing library
    local okDraw, _ = pcall(function() return type(Drawing) == "table" and Drawing.new end)
    caps.hasDrawing = okDraw

    return caps
end

Executor.Capabilities = nil
Executor.UNCScore = 0
Executor.UNCTotal = 20

function Executor.init()
    Executor.Capabilities = Executor.detectCapabilities()
    local count = 0
    for _, v in pairs(Executor.Capabilities) do
        if v then count = count + 1 end
    end
    Executor.UNCScore = count
    return Executor.Capabilities
end

-- ── Closure Analysis (getupvalues / getconstants / getprotos) ───────────────
-- These are the GC-based fallback tools. When decompile() fails completely,
-- we can still extract variable names/values and string constants from closures.

function Executor.getUpvalues(fn)
    -- Try UNC getupvalues
    local resolver = Executor.resolve("getupvalues") or Executor.resolve("debug.getupvalues")
    if resolver then
        local ok, upvals = pcall(resolver, fn)
        if ok and upvals then return upvals end
    end
    -- Try debug.getupvalue iteration
    local results = {}
    local idx = 1
    while true do
        local ok, name, val = pcall(debug.getupvalue, fn, idx)
        if not ok or name == nil then break end
        results[idx] = { name = name, value = val }
        idx = idx + 1
    end
    if #results > 0 then return results end
    return nil
end

-- ── GC Script Finder ────────────────────────────────────────────────────────
-- Scans getgc() for functions that belong to a specific script instance.
-- Returns all closures associated with that script.

function Executor.findScriptClosures(scriptInstance)
    local gc = Executor.getGarbageCollector(false)
    if #gc == 0 then return {} end

    local closures = {}
    local scriptPath = scriptInstance:GetFullName()

    for _, obj in ipairs(gc) do
        if type(obj) == "function" then
            local ok, env = pcall(getfenv, obj)
            if ok and env and type(env) == "table" then
                local envScript = rawget(env, "script")
                if envScript and typeof(envScript) == "Instance" then
                    if envScript == scriptInstance or envScript:GetFullName() == scriptPath then
                        table.insert(closures, obj)
                    end
                end
            end
        end
    end

    return closures
end

    State.selected = nil
    State.source = nil
    State.nodeCount = 0
    State.lastDecompileMethod = nil

function State.addFavorite(instance)
    local path = instance:GetFullName()
    for _, f in ipairs(State.favorites) do
        if f.path == path then return false end
    end
    table.insert(State.favorites, {
        name = instance.Name,
        path = path,
        class = instance.ClassName,
        hash = Executor.getScriptHash(instance),
        timestamp = os.time(),
    })
    State._shared.favorites = State.favorites
    return true
end

function State.removeFavorite(path)
    for i, f in ipairs(State.favorites) do
        if f.path == path then
            table.remove(State.favorites, i)
            State._shared.favorites = State.favorites
            return true
        end
    end
    return false
end

function State.isFavorite(instance)
    local path = instance:GetFullName()
    for _, f in ipairs(State.favorites) do
        if f.path == path then return true end
    end
    return false
end

function State.persistExpanded()
    State._shared.expanded = State.expanded
end

function State.logDecompile(scriptName, method, success, errorMsg)
    table.insert(State.decompileLog, {
        script = scriptName,
        method = method,
        success = success,
        error = errorMsg,
        time = os.time(),
    })
end

-- ╔═══════════════════════════════════════════════════════════════════════════╗
-- ║               MODULE: Decompiler — Robust 10-Method Chain                 ║
-- ║   decompile → closure → bytecode → getsenv → getgc+upvalues → require   ║
-- ║        Handles RunContext=Client, native code, and partial recovery       ║
-- ╚═══════════════════════════════════════════════════════════════════════════╝

local Decompiler = {}

-- ── Table Serializer ────────────────────────────────────────────────────────

local function serializeValue(value, depth, visited)
    depth = depth or 0
    visited = visited or {}

    if depth > 6 then return "{ --[[ depth limit ]] }" end

    local vType = type(value)

    if vType == "string" then
        local truncated = #value > 120 and value:sub(1, 120) .. "..." or value
        return '"' .. truncated:gsub('\\', '\\\\'):gsub('"', '\\"'):gsub("\n", "\\n"):gsub("\r", "\\r") .. '"'
    elseif vType == "number" or vType == "boolean" then
        return tostring(value)
    elseif vType == "nil" then
        return "nil"
    elseif vType == "function" then
        return "function() --[[ compiled ]] end"
    elseif vType == "table" then
        if visited[value] then return "{ --[[ circular ref ]] }" end
        visited[value] = true

        local indent = string.rep("    ", depth + 1)
        local closeIndent = string.rep("    ", depth)
        local parts = {}
        local count = 0

        for k, v in pairs(value) do
            count = count + 1
            if count > 30 then
                table.insert(parts, indent .. "-- ... " .. (next(value) and "more entries" or ""))
                break
            end

            local keyStr
            if type(k) == "string" and k:match("^[%a_][%w_]*$") then
                keyStr = k
            elseif type(k) == "string" then
                keyStr = '["' .. k .. '"]'
            else
                keyStr = "[" .. tostring(k) .. "]"
            end

            table.insert(parts, indent .. keyStr .. " = " .. serializeValue(v, depth + 1, visited))
        end

        if #parts == 0 then return "{}" end
        return "{\n" .. table.concat(parts, ",\n") .. "\n" .. closeIndent .. "}"
    else
        return "nil --[[ " .. vType .. ": " .. tostring(value) .. " ]]"
    end
end

-- ── Module Decompilation ────────────────────────────────────────────────────

function Decompiler.decompileModule(moduleScript)
    local ok, module = pcall(require, moduleScript)
    if not ok or module == nil then return nil end

    local result = "-- ╔══════════════════════════════════════════╗\n"
    result = result .. "-- ║  Decompiled ModuleScript (require())    ║\n"
    result = result .. "-- ╚══════════════════════════════════════════╝\n"
    result = result .. "-- Path: " .. moduleScript:GetFullName() .. "\n"
    result = result .. "-- Type: " .. type(module) .. "\n\n"

    if type(module) == "table" then
        result = result .. "return " .. serializeValue(module, 0)
    elseif type(module) == "function" then
        result = result .. "return function() --[[ compiled function ]] end"
    else
        result = result .. "return " .. tostring(module)
    end

    return result
end

-- ── GC-Based Closure Reconstruction ─────────────────────────────────────────
-- When decompile() completely fails, scan the garbage collector for functions
-- belonging to this script, then extract upvalues + constants to build
-- a partial reconstruction of the source.

function Decompiler.gcReconstruct(scriptInstance)
    -- Find all closures belonging to this script
    local closures = Executor.findScriptClosures(scriptInstance)
    if #closures == 0 then return nil end

    local result = "-- ╔══════════════════════════════════════════════════════╗\n"
    result = result .. "-- ║  GC Reconstruction (getgc + getupvalues)            ║\n"
    result = result .. "-- ║  Partial source recovered from garbage collector    ║\n"
    result = result .. "-- ╚══════════════════════════════════════════════════════╝\n"
    result = result .. "-- Script: " .. scriptInstance:GetFullName() .. "\n"
    result = result .. "-- Closures found: " .. #closures .. "\n\n"

    for i, closure in ipairs(closures) do
        if i > 20 then
            result = result .. "\n-- ... " .. (#closures - 20) .. " more closures omitted\n"
            break
        end

        result = result .. "-- ═══ Closure #" .. i .. " ═══════════════════════════════════\n"

        -- Try to decompile this individual closure
        local decompFn = Executor.resolve("decompile")
        if decompFn then
            local ok, src = pcall(decompFn, closure)
            if ok and src and type(src) == "string" and #src > 10 then
                result = result .. "-- [decompiled successfully]\n"
                result = result .. src .. "\n\n"
                -- Skip upvalue/constant dump for this one since we got real source
                continue
            end
        end

        -- Get upvalues
        local upvals = Executor.getUpvalues(closure)
        if upvals then
            result = result .. "-- Upvalues:\n"
            if type(upvals) == "table" then
                for idx, upval in pairs(upvals) do
                    if type(upval) == "table" and upval.name then
                        -- debug.getupvalue style: {name=..., value=...}
                        result = result .. "--   [" .. tostring(idx) .. "] " .. tostring(upval.name) .. " = "
                        if type(upval.value) == "table" then
                            result = result .. serializeValue(upval.value, 1)
                        else
                            result = result .. tostring(upval.value)
                        end
                        result = result .. "\n"
                    else
                        -- getupvalues style: {[idx] = value}
                        result = result .. "--   [" .. tostring(idx) .. "] = "
                        if type(upval) == "table" then
                            result = result .. serializeValue(upval, 1)
                        elseif type(upval) == "function" then
                            result = result .. "function() end"
                        else
                            result = result .. tostring(upval)
                        end
                        result = result .. "\n"
                    end
                end
            end
        end

        -- Get constants
        local consts = Executor.getConstants(closure)
        if consts then
            result = result .. "-- Constants:\n"
            local constCount = 0
            for idx, const in pairs(consts) do
                constCount = constCount + 1
                if constCount > 50 then
                    result = result .. "--   ... more constants\n"
                    break
                end
                result = result .. "--   [" .. tostring(idx) .. "] "
                if type(const) == "string" then
                    result = result .. '"' .. const:sub(1, 80) .. '"'
                else
                    result = result .. tostring(const)
                end
                result = result .. "\n"
            end
        end

        -- Get sub-protos (nested functions)
        local protos = Executor.getProtos(closure)
        if protos and #protos > 0 then
            result = result .. "-- Sub-functions: " .. #protos .. "\n"
            for pi, proto in ipairs(protos) do
                if pi > 5 then break end
                local protoConsts = Executor.getConstants(proto)
                if protoConsts then
                    result = result .. "--   Proto[" .. pi .. "] constants: "
                    local parts = {}
                    for _, c in pairs(protoConsts) do
                        if type(c) == "string" then
                            table.insert(parts, '"' .. c:sub(1, 40) .. '"')
                        end
                        if #parts >= 8 then break end
                    end
                    result = result .. table.concat(parts, ", ") .. "\n"
                end
            end
        end

        result = result .. "\n"
    end

    return result
end

-- ── Environment-Based Source Recovery ────────────────────────────────────────
-- Uses getsenv() to read all variables from a running script's environment.
-- This often works even when decompile() fails.

function Decompiler.envRecovery(scriptInstance)
    local env = Executor.getScriptEnvironment(scriptInstance)
    if not env then return nil end

    local result = "-- ╔══════════════════════════════════════════════════════╗\n"
    result = result .. "-- ║  Environment Recovery (getsenv)                     ║\n"
    result = result .. "-- ║  Live variables extracted from running script       ║\n"
    result = result .. "-- ╚══════════════════════════════════════════════════════╝\n"
    result = result .. "-- Script: " .. scriptInstance:GetFullName() .. "\n\n"

    local varCount = 0
    local functions = {}
    local tables = {}
    local values = {}

    for key, value in pairs(env) do
        if key ~= "script" and key ~= "_G" and key ~= "shared" then
            varCount = varCount + 1
            if varCount > 100 then
                result = result .. "-- ... (100+ variables, truncated)\n"
                break
            end

            if type(value) == "function" then
                table.insert(functions, key)
            elseif type(value) == "table" then
                table.insert(tables, { key = key, val = value })
            else
                table.insert(values, { key = key, val = value, vtype = type(value) })
            end
        end
    end

    -- Print simple values first
    if #values > 0 then
        result = result .. "-- ── Values ──────────────────────────\n"
        for _, v in ipairs(values) do
            if v.vtype == "string" then
                result = result .. "local " .. tostring(v.key) .. ' = "' .. tostring(v.val):sub(1, 80) .. '"\n'
            else
                result = result .. "local " .. tostring(v.key) .. " = " .. tostring(v.val) .. "\n"
            end
        end
        result = result .. "\n"
    end

    -- Print function names
    if #functions > 0 then
        result = result .. "-- ── Functions (" .. #functions .. ") ─────────────────────\n"
        for _, name in ipairs(functions) do
            result = result .. "local function " .. tostring(name) .. "(...) end  -- exists in env\n"

            -- Try to get upvalues of this function
            local fn = env[name]
            if fn and type(fn) == "function" then
                local upvals = Executor.getUpvalues(fn)
                if upvals then
                    for idx, uv in pairs(upvals) do
                        if type(uv) ~= "function" and type(uv) ~= "table" then
                            result = result .. "  -- upvalue[" .. tostring(idx) .. "] = " .. tostring(uv) .. "\n"
                        end
                    end
                end
            end
        end
        result = result .. "\n"
    end

    -- Print tables
    if #tables > 0 then
        result = result .. "-- ── Tables (" .. #tables .. ") ──────────────────────────\n"
        for _, t in ipairs(tables) do
            result = result .. "local " .. tostring(t.key) .. " = " .. serializeValue(t.val, 0) .. "\n\n"
        end
    end

    result = result .. "\n-- Total variables recovered: " .. varCount
    return result
end

-- ── RunContext Detection ────────────────────────────────────────────────────
-- Scripts with RunContext=Client appear as "Script" class but run client-side.
-- Many executors can't decompile these. We detect and handle them specially.

function Decompiler.isRunContextClient(scriptInstance)
    if not scriptInstance:IsA("Script") then return false end
    if scriptInstance:IsA("LocalScript") then return false end
    if scriptInstance:IsA("ModuleScript") then return false end

    -- It's a regular Script — check RunContext
    local ok, runContext = pcall(function()
        return scriptInstance.RunContext
    end)
    if ok and runContext then
        return runContext == Enum.RunContext.Client
    end

    -- Fallback: if a Script is in a client-side container, it might be RunContext=Client
    local path = scriptInstance:GetFullName():lower()
    if path:find("starterplayer") or path:find("startergui") or path:find("starterpack") then
        return true
    end

    return false
end

-- ══════════════════════════════════════════════════════════════════════════════
-- ═══ MAIN DECOMPILER CHAIN — 10 METHODS ═════════════════════════════════════
-- ══════════════════════════════════════════════════════════════════════════════
-- Each method is tried in order. The first one that returns >10 chars wins.
-- Every attempt is logged to State.decompileLog for debugging.

function Decompiler.getSource(scriptInstance)
    local log = {}
    local isRCC = Decompiler.isRunContextClient(scriptInstance)

    local methods = {
        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 1: Direct decompile (UNC standard)                      │
        -- │ Most executors with built-in decompilers support this.         │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "decompile(script)",
            fn = function()
                local decompFn = Executor.resolve("decompile")
                if not decompFn then return nil, "decompile not available" end
                local ok, src = pcall(decompFn, scriptInstance)
                if not ok then return nil, "pcall failed: " .. tostring(src) end
                if not src or type(src) ~= "string" then return nil, "returned non-string" end
                if #src < 10 then return nil, "returned too short (" .. #src .. " chars)" end
                -- Check for error messages disguised as source
                if src:find("^%-%-") and #src < 100 and src:find("fail") then
                    return nil, "returned error comment: " .. src:sub(1, 50)
                end
                return src
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 2: getscriptclosure → decompile(closure)                │
        -- │ Gets a fresh closure from bytecode, then decompiles it.        │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "closure→decompile",
            fn = function()
                local closure = Executor.getScriptClosure(scriptInstance)
                if not closure then return nil, "getscriptclosure failed" end
                local decompFn = Executor.resolve("decompile")
                if not decompFn then return nil, "decompile not available" end
                local ok, src = pcall(decompFn, closure)
                if not ok then return nil, "decompile(closure) failed: " .. tostring(src) end
                if not src or #src < 10 then return nil, "returned too short" end
                return src
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 3: gethiddenproperty("Source")                          │
        -- │ Reads the hidden Source property directly.                     │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "hiddenproperty:Source",
            fn = function()
                local fn = Executor.resolve("gethiddenproperty")
                if not fn then return nil, "gethiddenproperty not available" end
                local ok, src = pcall(fn, scriptInstance, "Source")
                if not ok then return nil, "pcall failed: " .. tostring(src) end
                if not src or type(src) ~= "string" or #src < 5 then return nil, "no source found" end
                return src
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 4: getscriptsource (legacy)                             │
        -- │ Older function, still works on some executors.                 │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "getscriptsource",
            fn = function()
                local fn = Executor.resolve("getscriptsource")
                if not fn then return nil, "getscriptsource not available" end
                local ok, src = pcall(fn, scriptInstance)
                if not ok then return nil, "pcall failed: " .. tostring(src) end
                if not src or #src < 5 then return nil, "no source returned" end
                return src
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 5: getsenv — environment variable recovery              │
        -- │ Gets all live variables from the running script's env.         │
        -- │ Works even on RunContext=Client scripts!                       │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "getsenv(env-recovery)",
            fn = function()
                return Decompiler.envRecovery(scriptInstance)
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 6: getgc + getupvalues + getconstants reconstruction    │
        -- │ Scans garbage collector for closures from this script,         │
        -- │ extracts their upvalues and constants for partial source.      │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "gc-reconstruct",
            fn = function()
                return Decompiler.gcReconstruct(scriptInstance)
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 7: require() for ModuleScripts                          │
        -- │ Runs the module and serializes its return value.               │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "module-require",
            fn = function()
                if not scriptInstance:IsA("ModuleScript") then
                    return nil, "not a ModuleScript"
                end
                return Decompiler.decompileModule(scriptInstance)
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 8: getscriptbytecode → bytecode dump                    │
        -- │ When nothing else works, dump raw bytecode as hex for manual   │
        -- │ analysis with external tools (unluau, luauDec).                │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "bytecode-dump",
            fn = function()
                local bytecode = Executor.getScriptBytecode(scriptInstance)
                if not bytecode or #bytecode < 4 then return nil, "no bytecode" end

                local result = "-- ╔══════════════════════════════════════════╗\n"
                result = result .. "-- ║  Bytecode Dump (for external decompiler) ║\n"
                result = result .. "-- ╚══════════════════════════════════════════╝\n"
                result = result .. "-- Script: " .. scriptInstance:GetFullName() .. "\n"
                result = result .. "-- Size: " .. #bytecode .. " bytes\n"
                result = result .. "-- Use with: unluau, luauDec, or similar tools\n\n"

                -- Show first 512 bytes as hex dump
                local maxBytes = math.min(#bytecode, 512)
                result = result .. "-- Hex dump (first " .. maxBytes .. " bytes):\n"
                for i = 1, maxBytes, 16 do
                    local hex = ""
                    local ascii = ""
                    for j = i, math.min(i + 15, maxBytes) do
                        local byte = string.byte(bytecode, j)
                        hex = hex .. string.format("%02X ", byte)
                        ascii = ascii .. (byte >= 32 and byte <= 126 and string.char(byte) or ".")
                    end
                    result = result .. string.format("-- %04X: %-48s | %s\n", i - 1, hex, ascii)
                end

                -- Extract readable strings from bytecode
                result = result .. "\n-- Extracted strings:\n"
                local strCount = 0
                for str in bytecode:gmatch("[%w_%.%:]+") do
                    if #str >= 4 and strCount < 60 then
                        strCount = strCount + 1
                        result = result .. '--   "' .. str:sub(1, 80) .. '"\n'
                    end
                end

                return result
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 9: getloadedmodules scan                                │
        -- │ Check if this module is in getloadedmodules and get its return │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "loadedmodules-scan",
            fn = function()
                if not scriptInstance:IsA("ModuleScript") then
                    return nil, "not a ModuleScript"
                end
                local modules = Executor.getLoadedModules()
                for _, mod in ipairs(modules) do
                    if mod == scriptInstance then
                        return Decompiler.decompileModule(scriptInstance)
                    end
                end
                return nil, "module not in getloadedmodules"
            end,
        },

        -- ┌─────────────────────────────────────────────────────────────────┐
        -- │ Method 10: getrunningscripts closure scan                      │
        -- │ Check if the script is in getrunningscripts, then try to       │
        -- │ get its closure from the running scripts list.                 │
        -- └─────────────────────────────────────────────────────────────────┘
        {
            name = "runningscripts-scan",
            fn = function()
                local running = Executor.getRunningScripts()
                for _, rs in ipairs(running) do
                    if rs == scriptInstance then
                        -- It's running, try getsenv on it
                        local envResult = Decompiler.envRecovery(scriptInstance)
                        if envResult then return envResult end

                        -- Try getting its closure via GC
                        local gcResult = Decompiler.gcReconstruct(scriptInstance)
                        if gcResult then return gcResult end

                        return nil, "found in running scripts but couldn't extract"
                    end
                end
                return nil, "not in getrunningscripts"
            end,
        },
    }

    -- ── Execute the chain ───────────────────────────────────────────────────
    for _, method in ipairs(methods) do
        local ok, result, errMsg = pcall(function()
            local src, err = method.fn()
            return src, err
        end)

        if ok and result and type(result) == "string" and #result > 10 then
            -- Count lines
            local lines = 1
            for _ in result:gmatch("\n") do lines = lines + 1 end

            -- Get hash if available
            local hash = Executor.getScriptHash(scriptInstance)

            -- Log success
            State.logDecompile(scriptInstance.Name, method.name, true, nil)

            return {
                source = result,
                lines = lines,
                bytes = #result,
                method = method.name,
                hash = hash,
                isRunContextClient = isRCC,
                log = log,
            }
        else
            local reason = "unknown"
            if not ok then
                reason = "error: " .. tostring(result)
            elseif errMsg then
                reason = errMsg
            elseif result then
                reason = "too short"
            else
                reason = "returned nil"
            end
            table.insert(log, method.name .. " → " .. reason)
            State.logDecompile(scriptInstance.Name, method.name, false, reason)
        end
    end

    -- ── All methods failed — generate detailed error report ─────────────────
    local info = "-- ╔══════════════════════════════════════════════════════════╗\n"
    info = info .. "-- ║  ⚠️  ALL DECOMPILATION METHODS FAILED                  ║\n"
    info = info .. "-- ╚══════════════════════════════════════════════════════════╝\n\n"
    info = info .. "-- Script: " .. scriptInstance.Name .. "\n"
    info = info .. "-- Class:  " .. scriptInstance.ClassName .. "\n"
    info = info .. "-- Path:   " .. scriptInstance:GetFullName() .. "\n"

    if isRCC then
        info = info .. "-- ⚠️  RunContext = Client (many executors cannot decompile these!)\n"
        info = info .. "--    Tip: This is a Server Script running as Client.\n"
        info = info .. "--    Only AWP.GG and a few executors support decompiling these.\n"
    end

    local hash = Executor.getScriptHash(scriptInstance)
    if hash then
        info = info .. "-- Hash:   " .. hash .. "\n"
    end

    info = info .. "-- Children: " .. #scriptInstance:GetChildren() .. "\n\n"

    info = info .. "-- ── Method Results ──────────────────────────────────────\n"
    for i, entry in ipairs(log) do
        info = info .. "-- " .. i .. ". " .. entry .. "\n"
    end

    info = info .. "\n-- ── Executor Capabilities ───────────────────────────────\n"
    info = info .. "-- UNC Score: " .. Executor.UNCScore .. "/" .. Executor.UNCTotal .. "\n"
    local caps = Executor.Capabilities or {}
    for k, v in pairs(caps) do
        if not v then
            info = info .. "-- ❌ " .. k .. "\n"
        end
    end

    info = info .. "\n-- ── What to try ─────────────────────────────────────────\n"
    info = info .. "-- 1. Use an executor with a built-in decompiler (100%% sUNC)\n"
    info = info .. "-- 2. Try saveinstance() to dump the whole game\n"
    info = info .. "-- 3. Use an external decompiler (unluau, luauDec)\n"
    if isRCC then
        info = info .. "-- 4. Use AWP.GG which handles RunContext=Client scripts\n"
    end

    return {
        source = info,
        lines = 30,
        bytes = #info,
        method = "ALL FAILED",
        hash = hash,
        isRunContextClient = isRCC,
        log = log,
    }
end

-- ── Full Game Map ───────────────────────────────────────────────────────────

function Scanner.generateMap()
    local maps = {}
    local totalObjs, totalScripts = 0, 0

    for _, serviceName in ipairs(Config.Services) do
        local ok, service = pcall(game.GetService, game, serviceName)
        if not ok or not service then continue end

        local lines, objs, scripts = {}, 0, 0

        local function scan(inst, depth, path)
            if objs > 2000 then return end
            objs = objs + 1
            if inst:IsA("BaseScript") then scripts = scripts + 1 end

            local icon = Scanner.getIcon(inst)
            local hash = ""
            if inst:IsA("BaseScript") and Config.Decompiler.ShowBytecodeHash then
                local h = Executor.getScriptHash(inst)
                if h then hash = " [hash:" .. h:sub(1, 12) .. "...]" end
            end

            table.insert(lines, string.rep("  ", depth) .. icon .. " " .. inst.Name .. " [" .. inst.ClassName .. "]" .. hash .. " -- " .. path)

            for _, child in ipairs(inst:GetChildren()) do
                pcall(scan, child, depth + 1, path .. "." .. child.Name)
            end
        end

        for _, child in ipairs(service:GetChildren()) do
            pcall(scan, child, 0, serviceName .. "." .. child.Name)
        end

        totalObjs = totalObjs + objs
        totalScripts = totalScripts + scripts
        maps[serviceName] = { lines = lines, objs = objs, scripts = scripts }
    end

    return maps, totalObjs, totalScripts
end

-- ── GC Object Scanner (UNC: getgc) ─────────────────────────────────────────
-- Scans garbage collector for functions, tables, and hidden references

function Scanner.scanGC(filterType)
    local gc = Executor.getGarbageCollector(true)
    local results = {}

    for _, obj in ipairs(gc) do
        if #results >= 200 then break end

        if filterType == nil or type(obj) == filterType then
            local info = { type = type(obj) }

            if type(obj) == "function" then
                pcall(function()
                    local dbg = debug.getinfo(obj)
                    info.name = dbg.name or "anonymous"
                    info.source = dbg.source or "unknown"
                    info.line = dbg.linedefined or 0
                end)
            elseif type(obj) == "table" then
                local count = 0
                for _ in pairs(obj) do count = count + 1 end
                info.size = count
            end

            table.insert(results, info)
        end
    end

    return results
end

-- ── Registry Scanner (UNC: getreg) ─────────────────────────────────────────

function Scanner.scanRegistry()
    local reg = Executor.getRegistry()
    local results = { functions = 0, tables = 0, threads = 0, other = 0 }

    for _, entry in ipairs(reg) do
        local t = type(entry)
        if t == "function" then results.functions = results.functions + 1
        elseif t == "table" then results.tables = results.tables + 1
        elseif t == "thread" then results.threads = results.threads + 1
        else results.other = results.other + 1 end
    end

    results.total = #reg
    return results
end

-- ── Search (Fuzzy Match Support) ────────────────────────────────────────────

function Scanner.search(query, parent)
    local results = {}
    query = query:lower()

    local function searchIn(instance)
        if #results >= Config.Search.MaxResults then return end

        for _, child in ipairs(instance:GetChildren()) do
            if #results >= Config.Search.MaxResults then return end

            local nameMatch = child.Name:lower():find(query, 1, true)
            local classMatch = child.ClassName:lower():find(query, 1, true)

            -- Fuzzy match: check if characters appear in order
            local fuzzyMatch = false
            if Config.Search.FuzzyMatch and not nameMatch and not classMatch then
                local name = child.Name:lower()
                local qi = 1
                for i = 1, #name do
                    if qi <= #query and name:sub(i, i) == query:sub(qi, qi) then
                        qi = qi + 1
                    end
                end
                fuzzyMatch = qi > #query
            end

            if nameMatch or classMatch or fuzzyMatch then
                table.insert(results, child)
            end

            pcall(searchIn, child)
        end
    end

    for _, serviceName in ipairs(Config.Services) do
        if #results >= Config.Search.MaxResults then break end
        local ok, service = pcall(game.GetService, game, serviceName)
        if ok and service then pcall(searchIn, service) end
    end

    return results
end

        local ok, service = pcall(game.GetService, game, serviceName)
        if not ok or not service then continue end

        local childCount = 0
        pcall(function() childCount = #service:GetChildren() end)

        order = order + 1

        local serviceNode = UI.frame({
            Name = "Service_" .. serviceName,
            Size = UDim2.new(1, 0, 0, Config.Tree.ItemHeight),
            Transparency = 1,
            Parent = State.treeScroll,
        })
        serviceNode.LayoutOrder = order
        serviceNode.AutomaticSize = Enum.AutomaticSize.Y

        local serviceBtn = Instance.new("TextButton")
        serviceBtn.Name = "ServiceBtn"
        serviceBtn.Size = UDim2.new(1, 0, 0, Config.Tree.ItemHeight)
        serviceBtn.BackgroundColor3 = Config.Colors.Service
        serviceBtn.BackgroundTransparency = 0.88
        serviceBtn.BorderSizePixel = 0
        serviceBtn.Font = Enum.Font.GothamBold
        serviceBtn.TextSize = 13
        serviceBtn.TextColor3 = Config.Colors.Service
        serviceBtn.TextXAlignment = Enum.TextXAlignment.Left
        serviceBtn.RichText = true
        serviceBtn.Text = '  ▸ 📂 ' .. serviceName .. ' <font color="#555">(' .. childCount .. ')</font>'
        serviceBtn.AutoButtonColor = false
        serviceBtn.Parent = serviceNode
        UI.corner(serviceBtn, 6)

        -- Service hover
        serviceBtn.MouseEnter:Connect(function()
            UI.tween(serviceBtn, {BackgroundTransparency = 0.75}, 0.1)
        end)
        serviceBtn.MouseLeave:Connect(function()
            UI.tween(serviceBtn, {BackgroundTransparency = 0.88}, 0.12)
        end)

        local serviceChildren = UI.frame({
            Name = "ServiceChildren",
            Size = UDim2.new(1, 0, 0, 0),
            Position = UDim2.new(0, 0, 0, Config.Tree.ItemHeight),
            Transparency = 1,
            Parent = serviceNode,
        })
        serviceChildren.Visible = false
        serviceChildren.AutomaticSize = Enum.AutomaticSize.Y
        UI.list(serviceChildren, 2)

        local serviceExpanded = false
        local serviceLoaded = false

        local function loadServiceChildren()
            if serviceLoaded then return end
            serviceLoaded = true

            local children = service:GetChildren()
            table.sort(children, function(a, b)
                local aScore = a:IsA("BaseScript") and 0 or (a:IsA("Folder") and 1 or 2)
                local bScore = b:IsA("BaseScript") and 0 or (b:IsA("Folder") and 1 or 2)
                if aScore == bScore then return a.Name < b.Name end
                return aScore < bScore
            end)

            for i, child in ipairs(children) do
                pcall(Explorer.createNode, child, serviceChildren, 1, i)
            end
        end

        local function updateServiceVisuals()
            serviceChildren.Visible = serviceExpanded
            local arrow = serviceExpanded and "▾" or "▸"
            serviceBtn.Text = '  ' .. arrow .. ' 📂 ' .. serviceName .. ' <font color="#555">(' .. childCount .. ')</font>'
        end

        -- Restore expanded state
        if State.expanded[serviceName] then
            serviceExpanded = true
            updateServiceVisuals()
            loadServiceChildren()
        end

        serviceBtn.MouseButton1Click:Connect(function()
            serviceExpanded = not serviceExpanded
            State.expanded[serviceName] = serviceExpanded or nil
            State.persistExpanded()
            updateServiceVisuals()
            if serviceExpanded then loadServiceChildren() end
        end)
    end
end

        Parent = codeHeader,
    })

    -- Meta labels
    State.metaLabel = UI.label({
        Text = "",
        Size = UDim2.new(1, -12, 0, 16),
        Position = UDim2.new(0, 6, 0, 46),
        Color = Config.Colors.Muted,
        TextSize = 11,
        Parent = codePanel,
    })

    State.hashLabel = UI.label({
        Text = "",
        Size = UDim2.new(1, -12, 0, 14),
        Position = UDim2.new(0, 6, 0, 62),
        Color = Config.Colors.TextSecondary,
        TextSize = 10,
        Mono = true,
        Parent = codePanel,
    })

    State.envLabel = UI.label({
        Text = "",
        Size = UDim2.new(1, -12, 0, 14),
        Position = UDim2.new(0, 6, 0, 76),
        Color = Config.Colors.ModuleScript,
        TextSize = 10,
        Parent = codePanel,
    })

    -- Code view
    local codeScroll = UI.scroll({
        Size = UDim2.new(1, 0, 1, -94),
        Position = UDim2.new(0, 0, 0, 94),
        Parent = codePanel,
    })

    State.codeLabel = Instance.new("TextBox")
    State.codeLabel.Name = "CodeView"
    State.codeLabel.Size = UDim2.new(1, -16, 0, 0)
    State.codeLabel.Position = UDim2.new(0, 8, 0, 8)
    State.codeLabel.BackgroundTransparency = 1
    State.codeLabel.Text = "-- Select a script to view source\n-- Use ▶ to execute, 📋 to copy, 🔬 to inspect env"
    State.codeLabel.TextColor3 = Config.Colors.Text
    State.codeLabel.Font = Enum.Font.Code
    State.codeLabel.TextSize = 13
    State.codeLabel.TextXAlignment = Enum.TextXAlignment.Left
    State.codeLabel.TextYAlignment = Enum.TextYAlignment.Top
    State.codeLabel.TextWrapped = false
    State.codeLabel.MultiLine = true
    State.codeLabel.ClearTextOnFocus = false
    State.codeLabel.TextEditable = false
    State.codeLabel.AutomaticSize = Enum.AutomaticSize.XY
    State.codeLabel.Parent = codeScroll

    -- ── Dragging ────────────────────────────────────────────────────────────
    local dragging, dragStart, startPos = false, nil, nil

    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = State.mainFrame.Position
        end
    end)

    header.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            State.mainFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    -- ── Build Tree & Animate In ─────────────────────────────────────────────
    Explorer.buildTree()

    -- Entrance animation
    State.mainFrame.Size = UDim2.new(0, 0, 0, 0)
    State.mainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
    task.wait(0.1)
    UI.tweenSpring(State.mainFrame, {
        Size = UDim2.fromScale(Config.Window.Width, Config.Window.Height),
        Position = UDim2.fromScale(0.5 - Config.Window.Width / 2, 0.5 - Config.Window.Height / 2),
    }, 0.4)

    -- ── Startup Toasts ──────────────────────────────────────────────────────
    task.wait(0.5)
    Toast.success("Script Explorer v" .. Config.Version .. " loaded!", 3)

    task.wait(0.3)
    local capCount = Executor.UNCScore
    if capCount >= 14 then
        Toast.info("🎯 UNC Score: " .. capCount .. "/" .. Executor.UNCTotal .. " — Excellent!", 4)
    elseif capCount >= 8 then
        Toast.warn("UNC Score: " .. capCount .. "/" .. Executor.UNCTotal .. " — Some features limited", 4)
    else
        Toast.error("UNC Score: " .. capCount .. "/" .. Executor.UNCTotal .. " — Many features unavailable", 5)
    end

    print("✅ Script Explorer v" .. Config.Version .. " [UNC " .. capCount .. "/" .. Executor.UNCTotal .. "] loaded!")
end

init()
